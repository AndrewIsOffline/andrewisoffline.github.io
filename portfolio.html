The syntax highlighting in this page is done by a WASM script compiled from a C++ source. Below is the source for that script.

#include <assert.h> #include <stdio.h> #include <stdlib.h> #include <string.h> // Typedefs #include "defs.hpp" // Structs for language keywords and symbols #include "keywords.hpp" // Exposes malloc for JS __attribute__((export_name("waMalloc"))) extern "C" void* waMalloc(uint size) { void *ptr = malloc(size); assert(ptr); return ptr; } // Exposes free for JS __attribute__((export_name("waFree"))) extern "C" void waFree(void *ptr) { free(ptr); } // ----------========== Char buffer class ==========---------- // Number of characters to be allocated at once #define NODESIZE 64 // Linked list node typedef struct CharNode_t { struct CharNode_t *next; // The next node in the list char data[NODESIZE]; // The data in the node } CharNode; class CharBuffer { CharNode *head, *tail; // List head and tail nodes uint size, pos; // Total character length and tail node fill amount void addTail(); public: CharBuffer(); void addChar(char ch); void addArray(const char *arr, uint len); void addString(const char *str); char* getString(); }; /* * addTail - Appends a node * Precondition: The current tail node is full * Behavior: Adds an empty node to the end of the data list */ inline void CharBuffer::addTail() { CharNode *n = (CharNode*)malloc(sizeof(CharNode)); assert(n); tail->next = n; tail = n; tail->next = NULL; pos = 0; } /* * CharBuffer - Constructor * Behavior: Initializes an empty buffer */ CharBuffer::CharBuffer() { head = (CharNode*)malloc(sizeof(CharNode)); assert(head); tail = head; tail->next = NULL; size = 0; pos = 0; } /* * addChar - Adds a byte * Parameters: A byte * Behavior: Adds the given byte to the end of the buffer */ void CharBuffer::addChar(char ch) { assert(pos < NODESIZE); tail->data[pos++] = ch; size++; if(pos >= NODESIZE) addTail(); } /* * addArray - Adds an array of bytes * Parameters: A pointer to an array of bytes and the length of that array * Behavior: Adds the bytes in the array to the end of the buffer */ void CharBuffer::addArray(const char *arr, uint len) { if(!len) return; assert(arr); assert(pos < NODESIZE); while(len + pos >= NODESIZE) { uint copylen = NODESIZE - pos; memcpy(&tail->data[pos], arr, copylen); len -= copylen; arr += copylen; size += copylen; addTail(); } if(!len) return; memcpy(&tail->data[pos], arr, len); pos += len; if(pos >= NODESIZE) addTail(); size += len; } /* * addString - Adds an string * Parameters: A pointer to a null-terminated string * Behavior: Adds the string to the end of the buffer */ void CharBuffer::addString(const char *str) { assert(str); addArray(str, strlen(str)); } /* * getString - Gets the data in the buffer * Output: A pointer to an array containing the data that was originally in the buffer * Behavior: Allocates a new array for output, frees any storage used by the buffer */ char* CharBuffer::getString() { char *out = (char*)malloc(size + 4); // Encode the length in little-endian order out[0] = (char)(size & 0xFF); out[1] = (char)((size >> 8) & 0xFF); out[2] = (char)((size >> 16) & 0xFF); out[3] = (char)((size >> 24) & 0xFF); uint outpos = 4; CharNode *npos = head; while(size > NODESIZE) { assert(npos); memcpy(&out[outpos], npos->data, NODESIZE); outpos += NODESIZE; size -= NODESIZE; CharNode *nnpos = npos->next; free(npos); npos = nnpos; } assert(npos); assert(!npos->next); memcpy(&out[outpos], npos->data, size); free(npos); return out; } // ----------========== Highlighting ==========---------- /* * writeTo - Writes characters to a buffer until a certain character is found * Parameters: The buffer to write to, the string to write, and the character to search * for * Output: The address of the character immediately after the one found * Behavior: Writes the first character in the string to the buffer, then keeps writing * characters until one matching the searched-for character is found, while properly * handling slashes and greater than/less than signs */ inline const char* writeTo(CharBuffer *buffer, const char* str, char ch) { buffer->addChar(*str); str++; bool skip = false; while(true) { char chat = *str; assert(chat); str++; if(chat == '<' || chat == '>') { buffer->addArray(chat == '<' ? "<" : ">", 4); } else buffer->addChar(chat); if(skip) { skip = false; } else if(chat == '\\') { skip = true; } else if(chat == ch) return str; } } // Enum for code types // MUST MATCH the enum in code.js! enum CodeType { CPPTYPE = 1, JAVATYPE = 2, ASMTYPE = 3 }; /* * highlight - Adds HTML highlighting * Parameters: An input string and a code type * Output: A copy of the input string, with HTML tags inserted to highlight the string * based on the code type given * Behavior: Allocates a new array for output */ __attribute__((export_name("highlight"))) extern "C" char* highlight(const char *text, CodeType type) { const MatchLibrary *library = NULL; switch(type) { case CPPTYPE: library = &CPPLIBRARY; break; case JAVATYPE: library = &JAVALIBRARY; break; case ASMTYPE: library = &ASMLIBRARY; break; default: // For some reason, without this printf the switch statement will start // throwing aborts even when a valid type is given printf("Invalid code type %i!\n", type); abort(); } CharBuffer buffer = CharBuffer(); const char *tpos = text; const uint32_t *lookup = library->symbols; while(*tpos) { // Find word const char *epos = tpos; while(true) { char chat = *epos; if(lookup[chat / 32] & (1 << (chat % 32))) break; epos++; } uint wlen = epos - tpos; char chat = *tpos; if(chat == library->spsymbol) { // Found string starting with a special character, like "#" in C buffer.addArray("<span class=\"cspecial\">", 23); buffer.addArray(tpos, wlen); buffer.addArray("</span>", 7); } else if(chat < 32 || chat > 127) { // First character is outside the bounds of the word array, just add as-is buffer.addArray(tpos, wlen); } else if(chat > 47 && chat < 58) { // Found number buffer.addArray("<span class=\"cnumber\">", 22); buffer.addArray(tpos, wlen); buffer.addArray("</span>", 7); } else { // Not a special case, look up the word in the list MatchEntry *entries = library->entries[chat - 32]; uint enpos; for(enpos = 0; enpos < library->llen[chat - 32]; enpos++) { if(entries[enpos].strlen != wlen) continue; if(!memcmp(tpos, entries[enpos].str, wlen)) break; } if(enpos == library->llen[chat - 32]) { // Miss, Just add the word as-is buffer.addArray(tpos, wlen); } else { // Hit, add formatting buffer.addArray("<span class=\"", 13); buffer.addString(CLASSNAMES[entries[enpos].style]); buffer.addArray("\">", 2); buffer.addArray(tpos, wlen); buffer.addArray("</span>", 7); } } tpos = epos; // Now scanning symbols while(true) { char chat = *tpos; if(!chat) goto endL; if(!(lookup[chat / 32] & (1 << (chat % 32)))) break; if(chat == '"' || chat == '\'') { // Found string buffer.addArray("<span class=\"cquote\">", 21); tpos = writeTo(&buffer, tpos, chat); buffer.addArray("</span>", 7); continue; } else if(chat == library->cmt[0] && !memcmp(tpos, library->cmt, library->cmtl)) { // Found end-of-line comment buffer.addArray("<span class=\"ccomment\">", 23); tpos = writeTo(&buffer, tpos, '\n'); buffer.addArray("</span>", 7); continue; } else if(chat == library->mlcs[0] && !memcmp(tpos, library->mlcs, library->mlcsl)) { // Found multi-line comment const char *eqpos = tpos + 1; eqpos = strstr(eqpos, library->mlce); assert(eqpos); eqpos += library->mlcel; buffer.addArray("<span class=\"ccomment\">", 23); for(const char *expos = tpos; expos < eqpos; expos++) { char chat = *expos; if(chat == '<' || chat == '>') { buffer.addArray(chat == '<' ? "<" : ">", 4); } else buffer.addChar(chat); } buffer.addArray("</span>", 7); tpos = eqpos; continue; } else if(chat == '<') { // Found <, change to code buffer.addArray("<", 4); tpos++; continue; } else if(chat == '>') { // Found >, change to code buffer.addArray(">", 4); tpos++; continue; } buffer.addChar(chat); tpos++; } } endL: buffer.addChar(0); return buffer.getString(); }
